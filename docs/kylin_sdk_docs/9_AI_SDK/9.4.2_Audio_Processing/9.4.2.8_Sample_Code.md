```c
// 实时语音识别示例
#include <coreai/speech/recognition.h>
#include <gio/gio.h>
#include <stdio.h>
void realtime_recognition_callback(SpeechRecognitionResult* result, void* user_data) {
    if (speech_recognition_result_get_reason(result) == SpeechResultReason_Recognizing) {
        printf("中间结果: %s\n", speech_recognition_result_get_text(result));
    } else if (speech_recognition_result_get_reason(result) == SpeechResultReason_Recognized) {
        printf("最终结果: %s\n", speech_recognition_result_get_text(result));
    }
}

```
